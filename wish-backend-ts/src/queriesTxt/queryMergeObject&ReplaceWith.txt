db.residents.aggregate([

    { $project: {"_id":1, email: 1, "costs.id": 1,  "costs.totalAnual": 1  } },
    
    { $unwind: { path: "$costs", preserveNullAndEmptyArrays: true } },
    
    {
        $lookup: {
           from: "payments",
           let: { resident_id: "$email", costId: "$costs.id" },
           pipeline: [
              { $match:
                 { $expr:
                    { $and:
                       [
                         { $eq: [ "$residentId",  "$$resident_id" ] },
                         { $eq: [ "$costId",  "$$costId" ] }
                       ]
                    }
                 }
              },
              { $addFields: {year: {  $year: "$date" }} },
              { $project: {residentId: 0, _id: 0} }
           ],
           as: "payments"
        }
    },
    
    {
        $unwind: { path: "$payments", preserveNullAndEmptyArrays: true }
    },

    {
      $group:
        {
          _id: { 
              id: "$_id",
              email: "$email",
              costId: "$costs.id",
              totalAnual: "$costs.totalAnual",
              paymentYear: "$payments.year"
          }, 
          totalPaidByYear: { $sum: "$payments.value" },
          payments: { $push: "$payments.value" }
        }
     },
     
     {
          $replaceRoot: { 
              newRoot: { 
                  $mergeObjects: ["$_id", { totalPaidByYear: "$totalPaidByYear", payments: "$payments"  }  ] 
              } 
          }
     },
     
     
     {
      $group:
        {
          _id: { 
              id: "$id",
              email: "$email",
              costId: "$costId"
          }, 
          years: { $push: {
                    year: "$paymentYear", 
                    totalAnual: "$totalAnual",
                    totalPaidByYear: "$totalPaidByYear",
                    payments: "$payments"
                }
              }
        }
     },
     
     {
          $replaceRoot: { 
              newRoot: { 
                  $mergeObjects: ["$_id", { years: "$years" }  ] 
              } 
          }
     },
     
     {
      $group:
        {
          _id: { 
              id: "$id",
              email: "$email",
          }, 
          costs: { $push: {
                    costId: "$costId",
                    years: "$years",
                }
              }
        }
     },
     
     {
          $replaceWith: {  $mergeObjects:  ["$_id", { costs: "$costs" }  ]  }
     },
     
     {
        $sort: { "_id.email": 1  }
     }
    
])
