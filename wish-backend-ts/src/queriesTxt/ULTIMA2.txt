  let collection = getDb().collection('residents');

    return await collection.aggregate([
        {
            '$unwind': {
                'path': '$costs',
                'preserveNullAndEmptyArrays': true
            }
        }, {
            '$lookup': {
                'from': 'recurrentCosts',
                'let': {
                    'residentId': '$_id',
                    'costId': '$costs.id'
                },
                'pipeline': [
                    {
                        '$match': {
                            '$expr': {
                                '$and': [
                                    {
                                        '$eq': [
                                            '$residentId', '$$residentId'
                                        ]
                                    }, {
                                        '$eq': [
                                            '$costId', '$$costId'
                                        ]
                                    }
                                ]
                            }
                        }
                    }, {
                        '$project': {
                            '_id': 0,
                            'resident': 0,
                            'costId': 0
                        }
                    }
                ],
                'as': 'recurrentRanges'
            }
        }, {
            '$set': {
                'recurrentRanges': {
                    '$first': '$recurrentRanges'
                }
            }
        }, {
            '$set': {
                'recurrentRanges': '$recurrentRanges.costRanges'
            }
        }, {
            '$lookup': {
                'from': 'payments',
                'let': {
                    'resResidentId': '$_id',
                    'resCostId': '$costs.id'
                },
                'pipeline': [
                    {
                        '$match': {
                            '$expr': {
                                '$and': [
                                    {
                                        '$eq': [
                                            '$residentId', '$$resResidentId'
                                        ]
                                    }, {
                                        '$eq': [
                                            '$costId', '$$resCostId'
                                        ]
                                    }, {
                                        '$gte': [
                                            {
                                                '$year': '$date'
                                            }, 2018
                                        ]
                                    }
                                ]
                            }
                        }
                    }, {
                        '$project': {
                            'residentId': 0,
                            '_id': 0
                        }
                    }
                ],
                'as': 'payments'
            }
        }, {
            '$group': {
                '_id': '$_id',
                'email': {
                    '$first': '$email'
                },
                'name': {
                    '$first': '$name'
                },
                'address': {
                    '$first': '$address'
                },
                'phoneNumber': {
                    '$first': '$phoneNumber'
                },
                'costs': {
                    '$push': {
                        'id': '$costs.id',
                        'name': '$costs.name',
                        'periodicity': '$costs.periodicity',
                        'totalAnual': "$costs.totalAnual",
                        'date': '$costs.date',
                        'recurrentRanges': '$recurrentRanges',
                        'payments': '$payments'
                    }
                }
            }
        }
    ])
        .toArray();
